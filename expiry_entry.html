CREATE TABLE IF NOT EXISTS expiry_entries (
    id BIGSERIAL PRIMARY KEY,
    code VARCHAR(50) NOT NULL,
    description TEXT NOT NULL,
    item_class VARCHAR(50) DEFAULT 'MDCN',
    unit_price NUMERIC(12, 5) DEFAULT 0,
    location VARCHAR(50) DEFAULT 'P0066',
    expiry_date VARCHAR(50) NOT NULL,
    quantity INTEGER DEFAULT 1,
    total_value NUMERIC(12, 2) DEFAULT 0,
    entry_date VARCHAR(50) NOT NULL,
    image_url TEXT,
    user_id VARCHAR(50) DEFAULT 'User 66',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable RLS and public access policy if needed
